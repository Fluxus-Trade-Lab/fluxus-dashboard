# 技能操作系统 v1 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 skill 在对的时刻被自动看见、用完必须留痕、裁决回流进菜谱；现存 79 个 skill/agent 全部过四档处置；三个首批住户挂上。

**Architecture:** 一张路由表 `SKILL_ROUTER.md` 是菜单；hook 甲（UserPromptSubmit）每条消息把菜单塞进上下文；hook 乙（Stop）收工前查 `skill-used/skipped/none` 行，没有就打回。两个 hook 都是无依赖的 Python 脚本，放 `.claude/hooks/`，由**提交进仓库的** `.claude/settings.json` 接线，本机交互、本机定时、云端 routine 三种会话同时生效。迭代靠每个 skill 末尾的「裁决记录」节 + 周检数「7 天内有裁决的行数」。

**Tech Stack:** Python 3.11+（标准库，零第三方依赖——hook 在云端 runner 上也要能跑）· pytest · Claude Code hooks（`.claude/settings.json`）· Workflow fan-out（审计阶段）

## Global Constraints

- Spec：`docs/superpowers/specs/2026-09-04-skill-os-design.md`，所有判据以它为准。
- hook 甲**永不阻断**；hook 乙**只查留痕不查对错**；`stop_hook_active=true` 时 hook 乙必须放行（防死锁）。
- 路由表以 **skill 精确名**为键，不依赖第三方 description。
- 状态只有四个字：`用` `修` `退` `建`；改状态必须带一句理由。
- 「退」永不删文件：第三方目录移到 `~/.claude/skills/_retired/`，自建 agent 移到 `.claude/agents/_retired/`。
- 每个新脚本必须有**注入 bug 能红**的正面对照步骤——这个项目「闸对、测试对、没人调用」栽过三次。
- 所有 commit 走「直推 main 标准动作」（临时树基于 `origin/main`，只 add 指名文件，push 后 `git log origin/main -1` 核实）。commit 尾行：`Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`。
- 中文注释；代码里的规矩要写「为什么」（这个仓库的惯例，看 `pipeline/adapters/yahoo_budget.py` 的口吻）。

```python
# pipeline/tools/skill_inventory.py
"""把四个来源的 skill / agent / command / 定时任务书列成一张表。

审计范围如果等于「我记得的地方」，漏掉的就永远是孤儿。所以来源写死成
参数，默认值指向这台机器的四个真实位置；测试用临时目录造假来源。
"""
from __future__ import annotations

import argparse
import json
import os
import re
from pathlib import Path
from typing import Dict, List

_FM = re.compile(r"^---\s*\n(.*?)\n---", re.S)


def read_description(path: Path) -> str:
    """frontmatter 里的 description；没有就空串，不抛。"""
    try:
        m = _FM.match(path.read_text(errors="replace"))
    except OSError:
        return ""
    if not m:
        return ""
    for line in m.group(1).splitlines():
        if line.strip().startswith("description:"):
            return line.split(":", 1)[1].strip().strip('"').strip("'")
    return ""


def _row(name: str, source: str, path: Path) -> Dict:
    return {
        "name": name,
        "source": source,
        "path": str(path),
        "description": read_description(path),
        "mtime": int(path.stat().st_mtime) if path.exists() else 0,
    }


def _skill_dirs(root: Path) -> List[Path]:
    if not root.exists():
        return []
    out = []
    for d in sorted(root.iterdir()):
        if not d.is_dir() or d.name.startswith("_"):
            continue  # _retired/ 等
        if (d / "SKILL.md").exists():
            out.append(d)
    return out


def inventory(roots: Dict[str, Path]) -> List[Dict]:
    rows: List[Dict] = []
    proj = roots.get("project")
    if proj:
        for d in _skill_dirs(proj / "skills"):
            rows.append(_row(d.name, "project-skill", d / "SKILL.md"))
        for f in sorted((proj / "agents").glob("*.md")) if (proj / "agents").exists() else []:
            if not f.parent.name.startswith("_"):
                rows.append(_row(f.stem, "project-agent", f))
        for f in sorted((proj / "commands").glob("*.md")) if (proj / "commands").exists() else []:
            rows.append(_row(f.stem, "project-command", f))
    us = roots.get("user-skills")
    if us:
        for d in _skill_dirs(us):
            rows.append(_row(d.name, "user-skill", d / "SKILL.md"))
    ps = roots.get("plugin-skills")
    if ps:
        for d in _skill_dirs(ps):
            rows.append(_row(d.name, "plugin-skill", d / "SKILL.md"))
    sc = roots.get("scheduled")
    if sc:
        for d in _skill_dirs(sc):
            rows.append(_row(d.name, "scheduled-task", d / "SKILL.md"))
    return rows


DEFAULT_ROOTS = {
    "project": Path(".claude"),
    "user-skills": Path(os.path.expanduser("~/.claude/skills")),
    "plugin-skills": Path(os.path.expanduser(
        "~/.claude/plugins/cache/claude-plugins-official/superpowers/6.2.0/skills")),
    "scheduled": Path(os.path.expanduser("~/.claude/scheduled-tasks")),
}


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description="列出全部 skill/agent 来源")
    ap.add_argument("--json", action="store_true")
    a = ap.parse_args(argv)
    rows = inventory(DEFAULT_ROOTS)
    if a.json:
        print(json.dumps(rows, ensure_ascii=False, indent=1))
    else:
        for r in rows:
            print(f"{r['source']:<16}{r['name']:<36}{r['description'][:60]}")
        print(f"\n共 {len(rows)} 项")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
# pipeline/tests/test_skill_hooks.py
"""两个 hook：甲永不阻断，乙只查留痕。

测试直接 import 脚本文件（它们不在包里），用 importlib 按路径加载。
每个 hook 各有一条「注入 bug 必须红」的对照在计划里，这里的测试是它们的靶子。
"""
import importlib.util
import json
from datetime import date
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[2]


def _load(name):
    p = ROOT / ".claude" / "hooks" / f"{name}.py"
```

```python
#!/usr/bin/env python3
# .claude/hooks/skill_router_inject.py
"""hook 甲：每条消息进来时，把 SKILL_ROUTER.md 的精简版塞进上下文。

为什么要有它：09-04 盘点，62 个第三方 skill 绝大多数零使用、自建 agent
08-23 后零调用——不是没人写菜谱，是菜单没端到桌上。这个 hook 就是端菜单。

两条硬规矩：
1. **永不阻断。** 路由表不存在、解析失败、任何异常，都只注入一行警告。
   菜单坏了不能让厨房停工。
2. **只注入状态≠退的行。** 「修」「建」带 ⚠️；「上次裁决」超 30 天带 ⏳——
   ⏳ 是给主人看的催办，不是给模型看的。
"""
from __future__ import annotations

import json
import sys
from datetime import date, datetime
from pathlib import Path
from typing import Dict, List

ROUTER = Path(__file__).resolve().parents[2] / "SKILL_ROUTER.md"
STALE_DAYS = 30
MAX_ROWS = 40
COLS = ["情境", "skill", "主人", "状态", "上次裁决", "备注"]
FOOTER = ("本轮若用了任一 skill，收工前必须写一行 `skill-used: <名> · <裁决/无>`；"
          "明知该用而没用，写 `skill-skipped: <名> · <原因>`；"
          "无适用，写 `skill-none: 本轮无适用 skill`。")


def parse_rows(text: str) -> List[Dict[str, str]]:
    rows: List[Dict[str, str]] = []
    for line in text.splitlines():
        if not line.lstrip().startswith("|"):
            continue
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) < 5:
            continue
        if cells[0] == COLS[0] or all(set(c) <= set("-: ") for c in cells):
            continue
        cells = (cells + [""] * 6)[:6]
        rows.append(dict(zip(COLS, cells)))
    return rows


def _stale(s: str, today: date) -> bool:
    try:
        d = datetime.strptime(s, "%Y-%m-%d").date()
    except ValueError:
        return True
    return (today - d).days > STALE_DAYS


def render(rows: List[Dict[str, str]], today: date) -> List[str]:
    out: List[str] = []
    for r in rows:
        if r["状态"] == "退":
            continue
        flag = " ⚠️" if r["状态"] in ("修", "建") else ""
        if _stale(r["上次裁决"], today):
            flag += " ⏳30天无裁决"
        out.append(f"- {r['情境']} → `{r['skill']}`（{r['主人']}·{r['状态']}{flag}）")
        if len(out) >= MAX_ROWS:
            break
    return out


def build_context(router_path: Path, today: date) -> str:
    try:
        lines = render(parse_rows(router_path.read_text()), today)
        head = "【技能路由表】\n" + "\n".join(lines)
    except FileNotFoundError:
        head = "【技能路由表】⚠️ SKILL_ROUTER.md 不存在，本轮无路由。"
    except Exception as e:  # noqa: BLE001 -- 菜单坏了不能停厨房
        head = f"【技能路由表】⚠️ 解析失败：{type(e).__name__}。"
    return head + "\n" + FOOTER


def main() -> None:
    body = build_context(ROUTER, date.today())
    print(json.dumps({"hookSpecificOutput": {
        "hookEventName": "UserPromptSubmit", "additionalContext": body}},
        ensure_ascii=False))


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
# .claude/hooks/skill_stop_gate.py
"""hook 乙：收工前查最后一条正文有没有留痕。没有就打回。

它查的是三种行之一：
    skill-used: <名> · <裁决/无>
    skill-skipped: <名> · <原因>
    skill-none: 本轮无适用 skill

它不查 skill 用得对不对——那是裁决的事。它只保证「用没用」这件事被写下来，
因为这个项目栽过三次「闸对、测试对、没人调用」，缺的从来不是判断，是留痕。

三条放行规则（失败模式必须是放行，不是卡死会话）：
1. stop_hook_active=true：已经在补写循环里，再拦就是死锁。
2. transcript 读不到：不拦。
3. 留痕只认正文 text 块，不认 thinking——留痕是给人看的。
"""
from __future__ import annotations

import json
import re
import sys
from typing import Dict, Optional

MARK = re.compile(r"skill-(used|skipped|none):")
REASON = ("收工前补一行：`skill-used: <名> · <裁决/无>`，"
          "或 `skill-skipped: <名> · <原因>`，或 `skill-none: 本轮无适用 skill`。")


def last_assistant_text(transcript_path: str) -> Optional[str]:
    text: Optional[str] = None
    try:
        with open(transcript_path, encoding="utf-8", errors="replace") as f:
            for line in f:
                try:
                    obj = json.loads(line)
                except ValueError:
                    continue
                if obj.get("type") != "assistant":
                    continue
                content = (obj.get("message") or {}).get("content")
                if isinstance(content, str):
                    text = content
                elif isinstance(content, list):
                    text = "\n".join(
                        c.get("text", "") for c in content
                        if isinstance(c, dict) and c.get("type") == "text")
    except OSError:
        return None
    return text


def verdict(payload: Dict) -> Dict:
    if payload.get("stop_hook_active"):
        return {}
    text = last_assistant_text(str(payload.get("transcript_path", "")))
    if text is None:
        return {}
    if MARK.search(text):
        return {}
    return {"decision": "block", "reason": REASON}


def main() -> None:
    try:
        payload = json.load(sys.stdin)
    except ValueError:
        payload = {}
    print(json.dumps(verdict(payload), ensure_ascii=False))


if __name__ == "__main__":
    main()
```

```python
# pipeline/tools/skill_router_check.py
"""周检读数：SKILL_ROUTER.md 里 7 天内有裁决的行数。

这是技能操作系统唯一的生命体征。hook 装得再漂亮，这个数不涨就是死的——
spec §六原话。parse_rows 与 hook 甲同构但独立一份：pipeline 不依赖 .claude/。
"""
from __future__ import annotations

import argparse
import json
from datetime import date, datetime
from pathlib import Path
from typing import Dict, List

ROUTER = Path("SKILL_ROUTER.md")
COLS = ["情境", "skill", "主人", "状态", "上次裁决", "备注"]


def parse_rows(text: str) -> List[Dict[str, str]]:
    rows: List[Dict[str, str]] = []
    for line in text.splitlines():
        if not line.lstrip().startswith("|"):
            continue
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) < 5 or cells[0] == COLS[0] or all(set(c) <= set("-: ") for c in cells):
            continue
        rows.append(dict(zip(COLS, (cells + [""] * 6)[:6])))
    return rows


def _age(s: str, today: date):
    try:
        return (today - datetime.strptime(s, "%Y-%m-%d").date()).days
    except ValueError:
        return None


def summarize(rows: List[Dict[str, str]], today: date) -> Dict:
    active = [r for r in rows if r["状态"] != "退"]
    fresh = [r for r in active if (_age(r["上次裁决"], today) or 10**6) <= 7]
    stale = [r["skill"] for r in active if (_age(r["上次裁决"], today) or 10**6) > 30]
    return {"total": len(rows), "active": len(active), "fresh_7d": len(fresh),
            "stale_30d": stale, "retired": len(rows) - len(active)}


def main(argv=None) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--json", action="store_true")
    ap.add_argument("--router", default=str(ROUTER))
    a = ap.parse_args(argv)
    s = summarize(parse_rows(Path(a.router).read_text()), date.today())
    if a.json:
        print(json.dumps(s, ensure_ascii=False))
    else:
        print(f"技能路由表：{s['active']} 行在用 · 7 天内有裁决 {s['fresh_7d']} 行 · "
              f"30 天无裁决 {len(s['stale_30d'])} 行 · 已退 {s['retired']}")
        for sk in s["stale_30d"]:
            print(f"  ⏳ {sk}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```markdown
# code-cartography — 先画地图，再动刀

> 方法不是想出来的，是 2026-09-04 那一轮跑出来的：108 个字段的产地盘点 → 消费图 →
> 实测推翻一条「这是 bug」的结论 → 注入 bug 验测试有分辨率 → 落地。这里把它写死。

## 什么时候用
- 周检（OPS 主人）：全仓扫 > 800 行的 py 和 > 300 行的 jsx，各出一份地图。
- 改 `frontend/src/components/` 之前（UI Claire 主人）：先给要改的那个组件画图。
- 有人说「这里乱了」——不许直接重构，先画图。

## 四步，缺一步就不算做过

### 1 · 盘消费者（谁在读它）
对目标文件的每个导出（函数 / 组件 / 字段），全仓 grep **至少三种拼法**
（驼峰、下划线、缩写、前端重命名），覆盖 `pipeline/` `frontend/src/` `data/reference/` 文档。
每个消费点写 `file:line + 一句话它拿这个干什么`。
判定四选一：`load_bearing` / `used_internally_only` / `computed_but_unused` / `duplicate_of`。
⚠️ **grep 干净 ≠ 没人用**；透传（`row[key]`、动态列配置）字面不出现，先找配置表。
⚠️ 一个文件若唯一调用者本身无人挂载，它就是死代码——顺手记下来，不顺手删。

### 2 · 量（不许凭感觉说乱）
至少三个数：行数 · 导出数 · 直接 fetch / useState / 全局状态的个数。
前端多量一个：这个组件的 props 有几个、其中几个只是往下透传。
后端多量一个：这个模块打了几个外部端点。

### 3 · 拆分建议（带 file:line，不带重写）
按「一起变的放一起」切，不按技术层切。每条建议写：
搬什么（file:line 范围）→ 搬到哪 → 谁会受影响（第 1 步的消费者清单）→ 怎么验证没坏。
**先量出「拆了省什么」**——09-04 那轮删 13 个字段省不下一次 Yahoo 请求，
这种话要在动手前说出来，不是动手后。

### 4 · 落地带正面对照
每个拆分 PR 必须：①既有测试全绿 ②新加一条守边界的测试 ③**注入一个真 bug 让它红一次**再还原。
红得不是地方（KeyError 式全红）不算。

## 不做什么
- 不做「顺手重构」——地图外的东西不碰。
- 不给建议配比喻，配 file:line。
- 不在一次里同时画图和动刀；图先合 main，刀另起分支。
```
